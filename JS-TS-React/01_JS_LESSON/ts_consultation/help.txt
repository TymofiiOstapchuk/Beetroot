============= install ts 

npm i --global typescript ts-node
tsc -v 

//---- init tsconfig.json
npx tsc --init


//---- compile 
npx tsc ./src/index.ts

//----	запуск файлу ts 
//---- https://github.com/TypeStrong/ts-node

npx ts-node ./src/index.ts
